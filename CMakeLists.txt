cmake_minimum_required(VERSION 3.20)
project(MindKeep VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(MINDKEEP_DEPS "" CACHE PATH "Optional locally extracted development packages")
if(MINDKEEP_DEPS)
  list(PREPEND CMAKE_PREFIX_PATH "${MINDKEEP_DEPS}/usr")
  include_directories("${MINDKEEP_DEPS}/usr/include" "${MINDKEEP_DEPS}/usr/include/x86_64-linux-gnu")
  link_directories("${MINDKEEP_DEPS}/usr/lib/x86_64-linux-gnu")
  find_library(MK_SYSTEM_CRYPTO NAMES libcrypto.so.3 PATHS /usr/lib/x86_64-linux-gnu /usr/lib64 NO_DEFAULT_PATH)
  if(MK_SYSTEM_CRYPTO)
    set(OPENSSL_CRYPTO_LIBRARY "${MK_SYSTEM_CRYPTO}" CACHE FILEPATH "Shared system crypto" FORCE)
  endif()
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ICU REQUIRED COMPONENTS uc i18n)
find_path(MYSQL_INCLUDE mysql.h PATH_SUFFIXES mysql REQUIRED)
find_library(MYSQL_LIB NAMES mysqlclient REQUIRED)
add_library(mindkeep_core backend/src/search.cpp backend/src/security.cpp)
target_include_directories(mindkeep_core PUBLIC backend/include vendor)
target_link_libraries(mindkeep_core PUBLIC OpenSSL::Crypto ICU::uc ICU::i18n Threads::Threads)
target_compile_options(mindkeep_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mindkeep backend/src/main.cpp backend/src/database.cpp backend/src/auth.cpp backend/src/notes.cpp backend/src/media.cpp backend/src/sharing.cpp backend/src/backup.cpp)
target_include_directories(mindkeep PRIVATE ${MYSQL_INCLUDE})
target_link_libraries(mindkeep PRIVATE mindkeep_core ${MYSQL_LIB})
target_compile_options(mindkeep PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(search_tests tests/search_tests.cpp)
target_link_libraries(search_tests PRIVATE mindkeep_core)
add_test(NAME search_correctness COMMAND search_tests)
add_executable(search_benchmark tests/search_benchmark.cpp)
target_link_libraries(search_benchmark PRIVATE mindkeep_core)
